% --- 問題の指示文 ---
\noindent\textbf{問題：} 次の二次関数のグラフと $x$ 軸の共有点の座標を求めよ。
\vspace{5mm}

% ==========================================
% multicols* でページ下端（bottom=30mmのライン）まで伸ばす
% ==========================================
\begin{multicols*}{2}
\begin{enumerate}[label=(\arabic*)]
    % --- 左列（1〜5問目） ---
    \item $y = x^2 - 4x + 3$
    
    {\color{red}
    $x^2 - 4x + 3 = 0$ とおくと
    
    $(x - 1)(x - 3) = 0$
    
    $x = 1, 3$
    
    \textbf{答.} $\boldsymbol{(1, 0), (3, 0)}$
    }
    \vspace*{\fill}

    \item $y = x^2 + 2x - 8$
    
    {\color{red}
    $x^2 + 2x - 8 = 0$ とおくと
    
    $(x + 4)(x - 2) = 0$
    
    $x = -4, 2$
    
    \textbf{答.} $\boldsymbol{(-4, 0), (2, 0)}$
    }
    \vspace*{\fill}

    \item $y = 2x^2 - 5x + 2$
    
    {\color{red}
    $2x^2 - 5x + 2 = 0$ とおくと
    
    $(2x - 1)(x - 2) = 0$
    
    $x = \frac{1}{2}, 2$
    
    \textbf{答.} $\boldsymbol{\left(\frac{1}{2}, 0\right), (2, 0)}$
    }
    \vspace*{\fill}

    \item $y = x^2 - 6x + 9$
    
    {\color{red}
    $x^2 - 6x + 9 = 0$ とおくと
    
    $(x - 3)^2 = 0$
    
    $x = 3$
    
    \textbf{答.} $\boldsymbol{(3, 0)}$
    }
    \vspace*{\fill}

    \item $y = -x^2 + 3x + 4$
    
    {\color{red}
    $-x^2 + 3x + 4 = 0$ とおくと
    
    $x^2 - 3x - 4 = 0$
    
    $(x - 4)(x + 1) = 0$
    
    $x = -1, 4$
    
    \textbf{答.} $\boldsymbol{(-1, 0), (4, 0)}$
    }
    \vspace*{\fill}
    
    \columnbreak

    % --- 右列（6〜10問目） ---
    \item $y = x^2 - 2x - 3$
    
    {\color{red}
    $x^2 - 2x - 3 = 0$ とおくと
    
    $(x - 3)(x + 1) = 0$
    
    $x = -1, 3$
    
    \textbf{答.} $\boldsymbol{(-1, 0), (3, 0)}$
    }
    \vspace*{\fill}

    \item $y = x^2 - 9$
    
    {\color{red}
    $x^2 - 9 = 0$ とおくと
    
    $(x - 3)(x + 3) = 0$
    
    $x = -3, 3$
    
    \textbf{答.} $\boldsymbol{(-3, 0), (3, 0)}$
    }
    \vspace*{\fill}

    \item $y = 3x^2 + 5x - 2$
    
    {\color{red}
    $3x^2 + 5x - 2 = 0$ とおくと
    
    $(3x - 1)(x + 2) = 0$
    
    $x = \frac{1}{3}, -2$
    
    \textbf{答.} $\boldsymbol{(-2, 0), \left(\frac{1}{3}, 0\right)}$
    }
    \vspace*{\fill}

    \item $y = -x^2 + 4x - 3$
    
    {\color{red}
    $-x^2 + 4x - 3 = 0$ とおくと
    
    $x^2 - 4x + 3 = 0$
    
    $(x - 1)(x - 3) = 0$
    
    $x = 1, 3$
    
    \textbf{答.} $\boldsymbol{(1, 0), (3, 0)}$
    }
    \vspace*{\fill}

    \item $y = x^2 + 4x + 4$
    
    {\color{red}
    $x^2 + 4x + 4 = 0$ とおくと
    
    $(x + 2)^2 = 0$
    
    $x = -2$
    
    \textbf{答.} $\boldsymbol{(-2, 0)}$
    }
    \vspace*{\fill}

\end{enumerate}
\end{multicols*}
